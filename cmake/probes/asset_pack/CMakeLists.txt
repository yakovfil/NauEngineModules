cmake_minimum_required(VERSION 3.24)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/.cmake/api/v1/query")
file(WRITE "${CMAKE_BINARY_DIR}/.cmake/api/v1/query/codemodel-v2" "")
get_filename_component(modules "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
get_filename_component(engine "${modules}/../NauEngineLite" ABSOLUTE)
set(NAU_ENGINE_SOURCE_DIR "${engine}")
if(CMAKE_TOOLCHAIN_FILE MATCHES "Emscripten\\.cmake$")
  include("${engine}/cmake/NauEmscriptenSdk.cmake")
  set(EMSCRIPTEN_SYSTEM_PROCESSOR wasm32 CACHE STRING "Target architecture")
endif()
project(NauAssetPackages LANGUAGES C CXX)
list(APPEND CMAKE_MODULE_PATH "${engine}/cmake")
include(NauCommon)
include(NauModule)
include(platformSpec/NauPlatformSetup)
set(NAU_RUNTIME_PROFILE minimal)
include("${engine}/cmake/NauStandaloneMinimal.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(NauEngineFolder NauAssetPackages)
include(NauGitVersion)
git_check_version()
file(STRINGS "${engine}/.nauversion" NAU_VERSION_STRING)
nau_set_version("${NAU_VERSION_STRING}")
set(NAU_MINIMAL_TESTS ON)
set(NAU_MINIMAL_GMOCK ON)
add_subdirectory("${engine}/engine/3rdparty_libs" dependencies)
add_subdirectory("${engine}/engine/core/kernel/src" kernel)
enable_testing()
add_executable(NauAssetPackTests "${engine}/engine/core/kernel/tests/test_kernel/test_io/test_asset_pack.cpp")
target_compile_definitions(NauAssetPackTests PRIVATE NAU_PACK_FIXTURES="${engine}/engine/core/kernel/tests/test_kernel/test_io/fixtures")
target_link_libraries(NauAssetPackTests PRIVATE NauKernel gtest_main)
nau_add_compile_options(NauAssetPackTests)
add_test(NAME AssetPackPortable COMMAND NauAssetPackTests)
if(EMSCRIPTEN)
  target_link_options(NauAssetPackTests PRIVATE -sEXIT_RUNTIME=1 -sPTHREAD_POOL_SIZE=8 -sALLOW_BLOCKING_ON_MAIN_THREAD=0
    "SHELL:--preload-file ${engine}/engine/core/kernel/tests/test_kernel/test_io/fixtures@${engine}/engine/core/kernel/tests/test_kernel/test_io/fixtures")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set_tests_properties(AssetPackPortable PROPERTIES DISABLED TRUE)
  add_test(NAME AssetPackChrome COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_browser.py" --build "${CMAKE_BINARY_DIR}")
endif()
target_sources(NauKernel PRIVATE "${engine}/engine/core/kernel/src/io/nau_container.cpp")
if(NOT EMSCRIPTEN)
  add_subdirectory("${modules}/tools/asset_pack_tool" pack_tool)
  add_executable(NauAssetPackBuilderTests "${modules}/tools/asset_pack_tool/tests/test_asset_pack_builder.cpp")
  target_link_libraries(NauAssetPackBuilderTests PRIVATE AssetPackTool gtest_main)
  nau_add_compile_options(NauAssetPackBuilderTests)
  add_test(NAME AssetPackBuilder COMMAND NauAssetPackBuilderTests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME AssetPackCLI COMMAND "${Python3_EXECUTABLE}" "${modules}/tools/asset_pack_tool/tests/test_cli.py"
    --tool $<TARGET_FILE:AssetPackToolCmd> --results "${CMAKE_BINARY_DIR}/cli-$<CONFIG>.json")
endif()
if(WIN32)
  target_sources(NauKernel PRIVATE
    "${engine}/engine/core/kernel/src/platform/windows/io/asset_pack_filesystem/asset_pack_file_system.cpp"
    "${engine}/engine/core/kernel/src/platform/windows/io/asset_pack_filesystem/asset_pack_file.cpp")
  add_executable(NauAssetPackLegacyTests "${engine}/engine/core/kernel/tests/test_kernel/test_io/test_asset_pack_legacy.cpp")
  target_compile_definitions(NauAssetPackLegacyTests PRIVATE NAU_PACK_FIXTURES="${engine}/engine/core/kernel/tests/test_kernel/test_io/fixtures")
  target_link_libraries(NauAssetPackLegacyTests PRIVATE NauKernel gtest_main)
  nau_add_compile_options(NauAssetPackLegacyTests)
  enable_testing()
  add_test(NAME AssetPackLegacy COMMAND NauAssetPackLegacyTests)
endif()
