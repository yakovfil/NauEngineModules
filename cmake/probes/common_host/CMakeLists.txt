cmake_minimum_required(VERSION 3.24)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/.cmake/api/v1/query")
file(WRITE "${CMAKE_BINARY_DIR}/.cmake/api/v1/query/codemodel-v2" "")
get_filename_component(modules "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
get_filename_component(engine "${modules}/../NauEngineLite" ABSOLUTE)
set(NAU_ENGINE_SOURCE_DIR "${engine}")
if(CMAKE_TOOLCHAIN_FILE MATCHES "Emscripten\\.cmake$" OR CMAKE_SYSTEM_NAME STREQUAL "Emscripten")
  include("${engine}/cmake/NauEmscriptenSdk.cmake")
  if(NOT EXISTS "$ENV{EMSDK_NODE}")
    message(FATAL_ERROR "EMSDK_NODE is missing; dot-source the active SDK emsdk_env.ps1 before lifecycle validation.")
  endif()
  set(CMAKE_CROSSCOMPILING_EMULATOR "$ENV{EMSDK_NODE}" CACHE STRING "Use the activated SDK Node runtime" FORCE)
  set(EMSCRIPTEN_SYSTEM_PROCESSOR wasm32 CACHE STRING "Target architecture")
endif()
project(NauCommonHost LANGUAGES C CXX)
list(APPEND CMAKE_MODULE_PATH "${engine}/cmake")
include(NauCommon)
include(NauModule)
include(platformSpec/NauPlatformSetup)
set(NAU_RUNTIME_PROFILE minimal)
set(NAU_MINIMAL_TESTS ON)
include("${modules}/cmake/NauMinimalRuntime.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(NauEngineFolder NauBrowserRuntime)
include(NauGitVersion)
git_check_version()
file(STRINGS "${engine}/.nauversion" NAU_VERSION_STRING)
nau_set_version("${NAU_VERSION_STRING}")
enable_testing()
set(NAU_MINIMAL_GMOCK ON)
add_subdirectory("${engine}/engine/3rdparty_libs" dependencies)
add_subdirectory("${engine}/engine/core/kernel/src" kernel)

add_subdirectory("${modules}/modules/platform_app/src" platform)
add_subdirectory("${engine}/engine/core/app_framework/src" framework)
add_executable(NauCommonHostTests "${engine}/engine/core/app_framework/tests/common_host/consumer.cpp")
target_link_libraries(NauCommonHostTests PRIVATE NauFrameworkMinimal)
nau_target_link_modules(NauCommonHostTests PlatformApp)
nau_add_compile_options(NauCommonHostTests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(EMSCRIPTEN)
  target_link_options(NauCommonHostTests PRIVATE -sEXIT_RUNTIME=1 -sPTHREAD_POOL_SIZE=12 -sALLOW_BLOCKING_ON_MAIN_THREAD=0)
endif()
foreach(scenario success legacy fixed immediate deferred stopstartup stoppre pending shutdownerror configfail setupfail runtime-pending runtime-legacy runtime-error runtime-startup-error runtime-cleanup-error)
  add_test(NAME CommonHost-${scenario} COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/run.py" --build "${CMAKE_BINARY_DIR}" --scenario ${scenario} --configuration $<CONFIG>)
  set_tests_properties(CommonHost-${scenario} PROPERTIES TIMEOUT 40 RUN_SERIAL TRUE)
endforeach()
if(EMSCRIPTEN)
  foreach(scenario stall visibility)
    add_test(NAME CommonHost-${scenario} COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/run.py" --build "${CMAKE_BINARY_DIR}" --scenario ${scenario} --configuration $<CONFIG>)
    set_tests_properties(CommonHost-${scenario} PROPERTIES TIMEOUT 40 RUN_SERIAL TRUE)
  endforeach()
else()
  add_executable(NauFileLoggingTests "${engine}/engine/core/app_framework/tests/common_host/test_file_logging.cpp")
  target_link_libraries(NauFileLoggingTests PRIVATE NauFrameworkMinimal gtest_main)
  nau_target_link_modules(NauFileLoggingTests PlatformApp)
  nau_add_compile_options(NauFileLoggingTests)
  add_test(NAME FileLoggingShutdown COMMAND NauFileLoggingTests)
  set(NAU_TEST_MODULES PlatformApp)
  add_subdirectory("${engine}/engine/core/app_framework/tests/test_minimal_app" lifecycle)
endif()
