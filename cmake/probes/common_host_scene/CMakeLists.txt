cmake_minimum_required(VERSION 3.24)
project(NauCommonHostScene LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(modules "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
get_filename_component(engine "${modules}/../NauEngineLite" ABSOLUTE)
set(NAU_MODULES_SOURCE_DIR "${modules}")
set(NAU_RUNTIME_PROFILE scene CACHE STRING "" FORCE)
set(NAU_EXTERNAL_RUNTIME_FILE "${CMAKE_CURRENT_LIST_DIR}/runtime.cmake" CACHE FILEPATH "" FORCE)
set(CMAKE_PROJECT_NauEngine_INCLUDE "${modules}/cmake/attach_modules.cmake" CACHE FILEPATH "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(NAU_CORE_TESTS OFF CACHE BOOL "" FORCE)
set(NAU_CORE_TOOLS OFF CACHE BOOL "" FORCE)
set(NAU_CORE_SAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${engine}" engine)
set(gtest_force_shared_crt OFF CACHE BOOL "" FORCE)
add_subdirectory("${engine}/engine/3rdparty_libs/googletest" googletest)
nau_target_add_compile_options(TARGETS gtest gtest_main)
add_executable(NauCommonHostSceneTests "${modules}/modules/scene/tests/test_scene/test_application_host.cpp")
target_link_libraries(NauCommonHostSceneTests PRIVATE NauFrameworkScene gtest_main)
nau_target_link_modules(NauCommonHostSceneTests CoreAssets CoreScene)
nau_add_compile_options(NauCommonHostSceneTests)
enable_testing()
add_test(NAME CommonHostScene COMMAND NauCommonHostSceneTests)
set_tests_properties(CommonHostScene PROPERTIES TIMEOUT 30)
