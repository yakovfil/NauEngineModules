cmake_minimum_required(VERSION 3.24)
project(NauWidgetConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF)
include(CTest)

if(NAU_WIDGET_SDK)
    # Import just the portable kernel dependency closure and the widget library.
    foreach(dependency IN ITEMS tinyutf8 fmt utf8cpp vectormath jsoncpp wyhash fast_float)
        include("${NAU_WIDGET_SDK}/cmake/3rdPartyLibs/${dependency}/${dependency}-config.cmake")
    endforeach()
    include("${NAU_WIDGET_SDK}/cmake/core/NauKernel/NauKernel-config.cmake")
    include("${NAU_WIDGET_SDK}/cmake/core/NauUi/NauUi-config.cmake")
    if(MSVC)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        add_compile_options(/GR-)
    endif()
else()
    if(NOT NAU_WIDGET_ENGINE)
        message(FATAL_ERROR "Set NAU_WIDGET_ENGINE to the standalone engine source or NAU_WIDGET_SDK to an installed SDK")
    endif()
    set(NAU_RUNTIME_PROFILE minimal CACHE STRING "" FORCE)
    set(NAU_CORE_TESTS OFF CACHE BOOL "" FORCE)
    set(NAU_CORE_TOOLS OFF CACHE BOOL "" FORCE)
    set(NAU_CORE_SAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory("${NAU_WIDGET_ENGINE}" engine)
    list(APPEND CMAKE_MODULE_PATH "${NAU_WIDGET_ENGINE}/cmake")
    include(NauCommon)
    include(NauModule)
    include(platformSpec/NauPlatformSetup)
    # The fixture explicitly opts into only the new feature module.
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." ui)
endif()

add_executable(NauWidgetConsumer main.cpp)
target_link_libraries(NauWidgetConsumer PRIVATE NauUi)
if(NOT NAU_WIDGET_SDK)
    nau_add_compile_options(NauWidgetConsumer)
endif()
if(EMSCRIPTEN)
    target_link_options(NauWidgetConsumer PRIVATE -sEXIT_RUNTIME=1)
    add_test(NAME WidgetConsumer COMMAND "$ENV{EMSDK_NODE}" $<TARGET_FILE:NauWidgetConsumer>)
else()
    add_test(NAME WidgetConsumer COMMAND NauWidgetConsumer)
endif()
set_tests_properties(WidgetConsumer PROPERTIES TIMEOUT 30)
